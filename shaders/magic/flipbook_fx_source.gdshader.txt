shader_type spatial;
render_mode unshaded, blend_add, cull_disabled, depth_draw_never;
// Plays an 8x8 video flipbook on a camera facing card. Black is invisible (additive).
uniform sampler2D sheet : source_color, filter_linear_mipmap;
uniform float columns = 8.0;
uniform float rows = 8.0;
uniform float frame = 0.0;          // set from script, 0 to 63, fractional blends frames
uniform float intensity = 1.6;      // HDR boost so glow and bloom pick it up
uniform float black_cut = 0.08;     // raise to remove leftover dim haze
uniform vec4 tint : source_color = vec4(1.0);
uniform float soft_depth = 0.6;     // fades where the card meets the ground
uniform sampler2D depth_tex : hint_depth_texture;
void vertex() {
	MODELVIEW_MATRIX = VIEW_MATRIX * mat4(INV_VIEW_MATRIX[0], INV_VIEW_MATRIX[1], INV_VIEW_MATRIX[2], MODEL_MATRIX[3]);
	MODELVIEW_MATRIX = MODELVIEW_MATRIX * mat4(vec4(length(MODEL_MATRIX[0].xyz),0,0,0), vec4(0,length(MODEL_MATRIX[1].xyz),0,0), vec4(0,0,length(MODEL_MATRIX[2].xyz),0), vec4(0,0,0,1));
}
vec3 cell(float f, vec2 uv) {
	float i = clamp(floor(f), 0.0, columns * rows - 1.0);
	vec2 c = vec2(mod(i, columns), floor(i / columns));
	return texture(sheet, (c + uv) / vec2(columns, rows)).rgb;
}
void fragment() {
	vec3 a = cell(frame, UV);
	vec3 b = cell(frame + 1.0, UV);
	vec3 col = mix(a, b, fract(frame));
	float l = max(col.r, max(col.g, col.b));
	col *= smoothstep(black_cut, black_cut + 0.08, l);
	float d = texture(depth_tex, SCREEN_UV).r;
	vec4 v = INV_PROJECTION_MATRIX * vec4(SCREEN_UV * 2.0 - 1.0, d, 1.0);
	float scene_z = -v.z / v.w;
	float fade = clamp((scene_z - (-VERTEX.z)) / soft_depth, 0.0, 1.0);
	ALBEDO = col * tint.rgb * intensity * fade;
}
